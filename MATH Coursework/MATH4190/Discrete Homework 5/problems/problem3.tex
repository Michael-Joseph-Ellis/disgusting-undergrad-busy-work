\section*{Problem 3. Use induction to prove each statement.}

\subsection*{(a) For $n\in\mathbb{N}$ with $n\ge 2$, we have $n!<n^n$.}
\textbf{Base case:} $n=2$: $2!=2<4=2^2$. \\
\textbf{Inductive step:} Assume $k!\!<\!k^k$ for some $k\ge 2$. Then
\[
(k+1)!=(k+1)\,k!<(k+1)\,k^k\le (k+1)\,(k+1)^k=(k+1)^{k+1},
\]
since $k^k\le (k+1)^k$. Hence $n!<n^n$ for all $n\ge2$. \qed

\subsection*{(b) For $n\in\mathbb{N}$ with $n\ge1$, $\displaystyle\sum_{i=1}^{n} i^{3}=\Big(\sum_{i=1}^{n} i\Big)^{2}$.}
\textbf{Base case:} $n=1$: $1^{3}=1=(1)^{2}$. \\
\textbf{Inductive step:} Assume $\sum_{i=1}^{k} i^{3}=\big(\sum_{i=1}^{k} i\big)^{2}=\big(\tfrac{k(k+1)}{2}\big)^{2}$. Then
\[
\sum_{i=1}^{k+1} i^{3}
=\bigg(\sum_{i=1}^{k} i^{3}\bigg)+(k+1)^{3}
=\bigg(\frac{k(k+1)}{2}\bigg)^{2}+(k+1)^{3}
=\bigg(\frac{(k+1)(k+2)}{2}\bigg)^{2}
=\bigg(\sum_{i=1}^{k+1} i\bigg)^{2}.
\]
Thus the identity holds for all $n\ge1$. \qed

\subsection*{(c) For all $n\in\mathbb{N}$, $5\mid \big(2^{\,2n+1}+3^{\,2n+1}\big)$.}
\textbf{Base case:} $n=0$: $2^{1}+3^{1}=5$, divisible by 5. \\
\textbf{Inductive step:} Assume $5\mid \big(2^{\,2k+1}+3^{\,2k+1}\big)$. Then
\[
2^{\,2(k+1)+1}+3^{\,2(k+1)+1}
=2^{\,2k+3}+3^{\,2k+3}
=4\cdot 2^{\,2k+1}+9\cdot 3^{\,2k+1}.
\]
Working modulo 5, note $4\equiv -1$ and $9\equiv -1$. Hence
\[
4\cdot 2^{\,2k+1}+9\cdot 3^{\,2k+1}\equiv
-\,\big(2^{\,2k+1}+3^{\,2k+1}\big)\equiv 0 \pmod{5},
\]
so $5$ divides $2^{\,2(k+1)+1}+3^{\,2(k+1)+1}$. By induction, the claim holds for all $n$. \qed

\subsection*{(d) Let $F_k$ be the $k$-th Fibonacci number. For every $k\in\mathbb{N}$, $F_{3k}$ is even.}
\textbf{Base case:} $k=1$: $F_{3}=2$ is even. \\
\textbf{Inductive step:} Assume for some $k\ge1$ that
\[
F_{3k}\ \text{is even},\qquad F_{3k+1}\ \text{is odd},\qquad F_{3k+2}\ \text{is odd}.
\]
Then using $F_{n+2}=F_{n+1}+F_n$,
\[
F_{3k+3}=F_{3k+2}+F_{3k+1}=\text{odd}+\text{odd}=\text{even}.
\]
Moreover,
\[
F_{3k+4}=F_{3k+3}+F_{3k+2}=\text{even}+\text{odd}=\text{odd},\quad
F_{3k+5}=F_{3k+4}+F_{3k+3}=\text{odd}+\text{even}=\text{odd}.
\]
Thus the parity pattern (even, odd, odd) repeats every three indices, so $F_{3(k+1)}$ is even. By induction, $F_{3k}$ is even for all $k\in\mathbb{N}$. \qed
