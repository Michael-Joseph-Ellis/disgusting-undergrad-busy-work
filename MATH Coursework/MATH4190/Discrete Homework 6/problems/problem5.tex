\section*{Problem 5 — Counting with generating functions}

\paragraph{Setup.}
We use ordinary generating functions (OGFs) for identical items and coefficient extraction $[z^m]F(z)$.

\subsection*{(i) 10 identical balloons to 4 children, each at least 2}
Each child has OGF $z^2/(1-z)$. Total OGF $\left(\dfrac{z^2}{1-z}\right)^4=z^8(1-z)^{-4}$.
We need $[z^{10}]$, i.e., $[z^{2}](1-z)^{-4}=\binom{2+4-1}{4-1}=\binom{5}{3}=10$.
\[
\boxed{10}
\]

\subsection*{(iv) Dozen bagels from egg, salty, plain; at least 2 of each, and at most 3 salty}
Egg and plain each: $z^2/(1-z)$; salty: $z^2+z^3$. Total OGF
\[
\frac{z^2}{1-z}\cdot\frac{z^2}{1-z}\cdot(z^2+z^3)=z^6\frac{1+z}{(1-z)^2}.
\]
We need $[z^{12}]$, i.e., $[z^{6}]\dfrac{1+z}{(1-z)^2}$.
Since $(1-z)^{-2}=\sum_{n\ge 0} (n+1)z^n$, we get
\[
[z^{6}]\frac{1+z}{(1-z)^2}=(6+1)+(5+1)=\boxed{13}.
\]
